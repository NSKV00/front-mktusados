import { useState } from 'react';
import { ArrowLeft, Loader2, MapPin } from 'lucide-react';
import { Button } from './ui/button';
import { Input } from './ui/input';
import { Label } from './ui/label';
import { Card } from './ui/card';
import { toast } from 'sonner@2.0.3';

interface AddressData {
  logradouro: string;
  bairro: string;
  cidade: string;
  estado: string;
}

export function AddressForm() {
  const [cep, setCep] = useState('');
  const [numero, setNumero] = useState('');
  const [addressData, setAddressData] = useState<AddressData | null>(null);
  const [isLoading, setIsLoading] = useState(false);
  const [cepError, setCepError] = useState('');

  // Format CEP with mask 00000-000
  const formatCEP = (value: string) => {
    const numbers = value.replace(/\D/g, '');
    if (numbers.length <= 5) {
      return numbers;
    }
    return `${numbers.slice(0, 5)}-${numbers.slice(5, 8)}`;
  };

  const handleCEPChange = (value: string) => {
    const formatted = formatCEP(value);
    setCep(formatted);
    setCepError('');
    
    // Clear address data if CEP is incomplete
    if (formatted.length < 9) {
      setAddressData(null);
    }
    
    // Trigger auto-fill when CEP is complete
    if (formatted.length === 9) {
      fetchAddressByCEP(formatted);
    }
  };

  // Mock CEP API lookup
  const fetchAddressByCEP = async (cepValue: string) => {
    setIsLoading(true);
    setCepError('');
    
    // Simulate API call
    await new Promise(resolve => setTimeout(resolve, 1000));
    
    const cepNumber = cepValue.replace(/\D/g, '');
    
    // Mock error for certain CEPs
    if (cepNumber === '00000000') {
      setCepError('CEP inválido. Por favor, verifique e tente novamente.');
      setAddressData(null);
      setIsLoading(false);
      return;
    }
    
    // Mock API failure for certain CEPs
    if (cepNumber === '99999999') {
      setCepError('Erro ao buscar CEP. Tente novamente mais tarde.');
      setAddressData(null);
      setIsLoading(false);
      return;
    }
    
    // Mock successful response
    const mockAddress: AddressData = {
      logradouro: 'Rua das Flores',
      bairro: 'Centro',
      cidade: 'São Paulo',
      estado: 'SP'
    };
    
    setAddressData(mockAddress);
    setIsLoading(false);
  };

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    
    // Validation
    if (!cep || cep.length !== 9) {
      toast.error('Por favor, insira um CEP válido.');
      return;
    }
    
    if (!numero) {
      toast.error('Por favor, insira o número.');
      return;
    }
    
    if (!addressData) {
      toast.error('Aguarde o preenchimento automático do endereço.');
      return;
    }
    
    // Mock successful save
    toast.success('Endereço cadastrado com sucesso!');
    
    // Reset form after success
    setTimeout(() => {
      setCep('');
      setNumero('');
      setAddressData(null);
    }, 2000);
  };

  const handleCancel = () => {
    setCep('');
    setNumero('');
    setAddressData(null);
    setCepError('');
    toast.info('Operação cancelada');
  };

  return (
    <div className="w-full min-h-screen px-4 py-6 md:flex md:items-center md:justify-center">
      <div className="w-full max-w-[600px] mx-auto">
        {/* Header */}
        <div className="flex items-center gap-4 mb-6">
          <button 
            className="p-2 -ml-2 hover:bg-gray-200 rounded-full transition-colors"
            aria-label="Voltar"
          >
            <ArrowLeft className="w-6 h-6 text-gray-700" />
          </button>
          <h1 className="text-gray-900">Adicionar Endereço</h1>
        </div>

        {/* Main Card */}
        <Card className="p-6 md:p-8 shadow-lg rounded-[20px] border-0">
          <h2 className="text-center text-gray-900 mb-6">Adicionar endereço</h2>
          
          <form onSubmit={handleSubmit} className="space-y-6">
            {/* CEP Input */}
            <div className="space-y-2">
              <Label htmlFor="cep">
                CEP <span className="text-red-500">*</span>
              </Label>
              <div className="relative">
                <Input
                  id="cep"
                  value={cep}
                  onChange={(e) => handleCEPChange(e.target.value)}
                  placeholder="00000-000"
                  maxLength={9}
                  className={cepError ? 'border-red-500 focus-visible:ring-red-500' : ''}
                />
                {isLoading && (
                  <Loader2 className="w-5 h-5 text-blue-500 animate-spin absolute right-3 top-1/2 -translate-y-1/2" />
                )}
              </div>
              {cepError ? (
                <p className="text-sm text-red-600">{cepError}</p>
              ) : (
                <p className="text-sm text-gray-500">
                  Digite o CEP para preencher automaticamente
                </p>
              )}
            </div>

            {/* Auto-filled Address Section */}
            {addressData && (
              <div className="bg-gray-50 rounded-lg p-4 space-y-4 border border-gray-200">
                <div className="flex items-center gap-2 mb-2">
                  <MapPin className="w-4 h-4 text-gray-600" />
                  <span className="text-sm text-gray-700">Endereço encontrado</span>
                </div>
                
                <div className="grid grid-cols-1 gap-4">
                  <div className="space-y-2">
                    <Label htmlFor="logradouro" className="text-gray-600">Logradouro</Label>
                    <Input
                      id="logradouro"
                      value={addressData.logradouro}
                      disabled
                      className="bg-white"
                    />
                  </div>

                  <div className="space-y-2">
                    <Label htmlFor="bairro" className="text-gray-600">Bairro</Label>
                    <Input
                      id="bairro"
                      value={addressData.bairro}
                      disabled
                      className="bg-white"
                    />
                  </div>

                  <div className="grid grid-cols-2 gap-4">
                    <div className="space-y-2">
                      <Label htmlFor="cidade" className="text-gray-600">Cidade</Label>
                      <Input
                        id="cidade"
                        value={addressData.cidade}
                        disabled
                        className="bg-white"
                      />
                    </div>

                    <div className="space-y-2">
                      <Label htmlFor="estado" className="text-gray-600">Estado</Label>
                      <Input
                        id="estado"
                        value={addressData.estado}
                        disabled
                        className="bg-white"
                      />
                    </div>
                  </div>
                </div>
              </div>
            )}

            {/* Número Input */}
            <div className="space-y-2">
              <Label htmlFor="numero">
                Número <span className="text-red-500">*</span>
              </Label>
              <Input
                id="numero"
                type="text"
                inputMode="numeric"
                value={numero}
                onChange={(e) => {
                  const value = e.target.value.replace(/\D/g, '');
                  setNumero(value);
                }}
                placeholder="123"
              />
            </div>

            {/* Cidade - Read Only */}
            <div className="space-y-2">
              <Label htmlFor="cidade-display" className="text-gray-600">Cidade</Label>
              <Input
                id="cidade-display"
                value={addressData?.cidade || ''}
                disabled
                className="bg-gray-50"
                placeholder="Será preenchido automaticamente"
              />
            </div>

            {/* Estado - Read Only */}
            <div className="space-y-2">
              <Label htmlFor="estado-display" className="text-gray-600">Estado</Label>
              <Input
                id="estado-display"
                value={addressData?.estado || ''}
                disabled
                className="bg-gray-50"
                placeholder="Será preenchido automaticamente"
              />
            </div>

            {/* Action Buttons */}
            <div className="pt-6 space-y-4">
              <Button
                type="submit"
                className="w-full h-12 bg-purple-600 hover:bg-purple-700 text-white rounded-xl"
              >
                Salvar Endereço
              </Button>

              <Button
                type="button"
                onClick={handleCancel}
                variant="secondary"
                className="w-full h-12 bg-gray-400 hover:bg-gray-500 text-white rounded-xl"
              >
                Cancelar
              </Button>
            </div>
          </form>
        </Card>

        {/* Helper Text */}
        <p className="text-center text-sm text-gray-500 mt-6">
          Os campos marcados com <span className="text-red-500">*</span> são obrigatórios
        </p>
      </div>
    </div>
  );
}
